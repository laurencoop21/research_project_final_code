import rasterio
import numpy as np

path = "Iraq_ND11_4_slope_TPI_uint8.tif"

# Approximate location you were just viewing in QGIS

x = 553999
y = 3545614

with rasterio.open(path) as src:
    row, col = src.index(x, y)

    # Print a small patch around that location

    tpi = src.read(
        3,
        window=rasterio.windows.Window(
            col - 6,
            row - 6,
            12,
            12,
        ),
    )

print("12 x 12 TPI values around the QGIS location")
print("--------------------------------------------")
print(tpi)

print("\nUnique values in this patch:", len(np.unique(tpi)))

# --------------------------------------------------

# Check a larger patch for suspicious constant 3x3 blocks

# --------------------------------------------------


with rasterio.open(path) as src:
    row, col = src.index(x, y)

    large = src.read(
        3,
        window=rasterio.windows.Window(
            col - 75,
            row - 75,
            150,
            150,
        ),
    )

print("\nConstant 3x3 block check")
print("------------------------")

for row_offset in range(3):
    for col_offset in range(3):

        total = 0
        constant = 0

        for r in range(row_offset, large.shape[0] - 2, 3):
            for c in range(col_offset, large.shape[1] - 2, 3):

                block = large[r:r+3, c:c+3]

                if block.shape != (3, 3):
                    continue

                total += 1

                if np.all(block == block[0, 0]):
                    constant += 1

        percentage = 100 * constant / total if total else 0

        print(
            f"offset ({row_offset},{col_offset}): "
            f"{constant}/{total} constant 3x3 blocks "
            f"({percentage:.1f}%)"
        )

================ HISTORY 356 ================
	 python - <<'PY'
import pandas as pd
import geopandas as gpd
import numpy as np

model = "../model_12_nd11_4_slope_tpi_newdata_yolo"

# --------------------------------------------------

# Load Model 12 unique-tell detections

# --------------------------------------------------


det = pd.read_csv(
    model + "/reports/model_12_unique_tell_detection.csv"
)

# --------------------------------------------------

# Load geographic split

# --------------------------------------------------


gpkg = model + "/reports/geographic_split.gpkg"

tells = gpd.read_file(
    gpkg,
    layer="tells_with_split",
)

regions = gpd.read_file(
    gpkg,
    layer="split_regions",
)

test_tells = tells[tells["split"] == "test"].copy()

# --------------------------------------------------

# Merge detections with test tell locations

# --------------------------------------------------


df = test_tells[
    ["polygon_id", "centroid_x"]
].merge(
    det,
    on="polygon_id",
    how="left",
    validate="one_to_one",
)

print("Matched test tells:", len(df))
print("Missing detection records:", df["detected"].isna().sum())

assert len(df) == 201
assert df["detected"].isna().sum() == 0

# --------------------------------------------------

# Get exact western/eastern bounds of test region

# --------------------------------------------------


test_region = regions[
    regions["split"] == "test"
].geometry.iloc[0]

west = test_region.bounds[0]
east = test_region.bounds[2]
width = east - west

df["distance_from_west_km"] = (
    df["centroid_x"] - west
) / 1000

df["relative_east_position"] = (
    df["centroid_x"] - west
) / width

# --------------------------------------------------

# Divide test strip into equal-width thirds

# --------------------------------------------------


df["test_third"] = pd.cut(
    df["relative_east_position"],
    bins=[-np.inf, 1/3, 2/3, np.inf],
    labels=[
        "western third",
        "middle third",
        "eastern third",
    ],
)

# --------------------------------------------------

# Overall recall by geographic third

# --------------------------------------------------


summary = (
    df.groupby("test_third", observed=False)
    .agg(
        tells=("polygon_id", "count"),
        detected=("detected", "sum"),
    )
)

summary["recall"] = (
    summary["detected"] / summary["tells"]
)

print("\nMODEL 12 UNIQUE-TELL RECALL BY TEST-REGION THIRD")
print("------------------------------------------------")
print(summary)

# --------------------------------------------------

# Tell size distribution in each third

# --------------------------------------------------


size_counts = pd.crosstab(
    df["test_third"],
    df["size_class"],
)

print("\nTELL SIZE COUNTS BY THIRD")
print("-------------------------")
print(size_counts)

# --------------------------------------------------

# Recall by size within each geographic third

# --------------------------------------------------


size_recall = (
    df.groupby(
        ["test_third", "size_class"],
        observed=False,
    )
    .agg(
        tells=("polygon_id", "count"),
        detected=("detected", "sum"),
    )
)

size_recall["recall"] = np.where(
    size_recall["tells"] > 0,
    size_recall["detected"] / size_recall["tells"],
    np.nan,
)

print("\nRECALL BY THIRD AND SIZE CLASS")
print("------------------------------")
print(size_recall)

# --------------------------------------------------

# Compare locations of detected vs missed tells

# --------------------------------------------------


location_summary = (
    df.groupby("detected")
    .agg(
        tells=("polygon_id", "count"),
        mean_distance_km=("distance_from_west_km", "mean"),
        median_distance_km=("distance_from_west_km", "median"),
        mean_relative_position=("relative_east_position", "mean"),
    )
)

print("\nDETECTED VS MISSED LOCATION")
print("---------------------------")
print(location_summary)
